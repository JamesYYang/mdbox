# 图片库（内容寻址 assets）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 Markdown 正文能引用本站图片——Web 端粘贴/拖拽/按钮上传，MCP 端 `upload_image`，图片按内容寻址存在 `data/users/<user>/assets/`，经 `/a/<user>/<hash>.<ext>` 匿名读取。

**Architecture:** 沿用「文件即真相」：图片是用户目录下的普通文件，内存里只多一张「文件名 → 字节数」的索引用于配额。写入只走 `store.SaveImage`（大小/类型/去重/配额四件事都在这里），REST 与 MCP 共用它；读取是一个独立于 `/api/` 的匿名路由，不碰 `withAuth` 与 `publicPath`。

**Tech Stack:** Go 1.23+（标准库 `crypto/sha256`、`net/http`、`mime/multipart`、`encoding/base64`）、goldmark/bluemonday（不改）、原生 JS（无框架、无新依赖）。

## Global Constraints

- 设计稿：`docs/superpowers/specs/2026-11-08-images-design.md`（本计划的唯一事实源，冲突时以设计稿为准）。
- **不新增任何第三方依赖**（WebP 只做类型识别，不做解码）。
- 注释与用户可见文案用中文；用户可见错误必须中英双语（`internal/i18n`）。
- 限额：单图 5 MiB（`store.MaxImageBytes`）、单篇正文 2 MiB、单用户总量 200 MiB（图片计入）。
- 允许的图片类型只有 `image/png`、`image/jpeg`、`image/gif`、`image/webp`；SVG 一律拒收。
- 图片 URL 必须是站点根绝对路径 `/a/<user>/<name>`，不要写相对路径。
- 前端零依赖、不用 CDN；`web/` 通过 `//go:embed` 打进二进制，改前端后必须重新 `go build`。
- 本机是 CRLF 检出，`gofmt -l .` 会把所有文件都列出来；判断格式只用 `gofmt -w <改过的文件>` 后看 `git diff`。
- 提交直接落在 `main`，不建分支、不推送（推送由作者确认后单独做）。
- 每个任务结束都要能独立通过：`go build -o /tmp/mdbox_check .` + 该任务的 `go test`。

```go
package store

import (
	"crypto/sha256"
	"encoding/hex"
	"errors"
	"net/http"
	"os"
	"path/filepath"
	"regexp"

	"mdbox/internal/i18n"
)

// MaxImageBytes 是单张图片上限。图片与文档共用 MaxUserBytes 配额。
const MaxImageBytes = 5 << 20

var (
	// ErrImageTooLarge 表示图片超过 MaxImageBytes。
	ErrImageTooLarge = errors.New(string(i18n.KeyImageTooLarge))
	// ErrUnsupportedImage 表示内容不是允许的图片格式（SVG 也走这里）。
	ErrUnsupportedImage = errors.New(string(i18n.KeyUnsupportedImage))
)

// assetNameRe 限定可读取的文件名：32 位小写 hex + 已知扩展名。
// 这个正则同时是路径穿越的防线，OpenImage 之外不要用它做别的判断。
var assetNameRe = regexp.MustCompile(`^[0-9a-f]{32}\.(png|jpg|gif|webp)$`)

// imageExt 把嗅探出的 MIME 映射成扩展名；不在白名单返回空串。
func imageExt(mime string) string {
	switch mime {
	case "image/png":
		return ".png"
	case "image/jpeg":
		return ".jpg"
	case "image/gif":
		return ".gif"
	case "image/webp":
		return ".webp"
	}
	return ""
}

// SaveImage 按内容存一张图片并返回文件名。
// 校验顺序：大小 → 类型；同名（同内容）已存在时直接复用，不再占配额。
func (s *Store) SaveImage(data []byte) (string, int64, error) {
	if len(data) > MaxImageBytes {
		return "", 0, ErrImageTooLarge
	}
	probe := data
	if len(probe) > 512 {
		probe = probe[:512]
	}
	ext := imageExt(http.DetectContentType(probe))
	if ext == "" {
		return "", 0, ErrUnsupportedImage
	}
	sum := sha256.Sum256(data)
	name := hex.EncodeToString(sum[:16]) + ext

	s.mu.Lock()
	defer s.mu.Unlock()
	if n, ok := s.assets[name]; ok {
		return name, n, nil
	}
	if s.usedBytes()+int64(len(data)) > s.quota {
		return "", 0, ErrQuotaExceeded
	}
	dir := filepath.Join(s.root, "assets")
	if err := os.MkdirAll(dir, 0o755); err != nil {
		return "", 0, err
	}
	tmp, err := os.CreateTemp(dir, "tmp-*")
	if err != nil {
		return "", 0, err
	}
	tmpName := tmp.Name()
	cleanup := func() {
		_ = tmp.Close()
		_ = os.Remove(tmpName)
	}
	if _, err := tmp.Write(data); err != nil {
		cleanup()
		return "", 0, err
	}
	if err := tmp.Close(); err != nil {
		_ = os.Remove(tmpName)
		return "", 0, err
	}
	// 先落盘再进索引：写失败时内存与磁盘不会脱节。
	if err := os.Rename(tmpName, filepath.Join(dir, name)); err != nil {
		_ = os.Remove(tmpName)
		return "", 0, err
	}
	s.assets[name] = int64(len(data))
	return name, int64(len(data)), nil
}

// OpenImage 按名打开图片。名字不合法或文件不存在都返回 ErrNotFound，
// 不区分「非法」与「不存在」，避免被用来探测。
func (s *Store) OpenImage(name string) (*os.File, os.FileInfo, error) {
	if !assetNameRe.MatchString(name) {
		return nil, nil, ErrNotFound
	}
	f, err := os.Open(filepath.Join(s.root, "assets", name))
	if err != nil {
		if os.IsNotExist(err) {
			return nil, nil, ErrNotFound
		}
		return nil, nil, err
	}
	fi, err := f.Stat()
	if err != nil {
		_ = f.Close()
		return nil, nil, err
	}
	return f, fi, nil
}
```

func (s *Store) usedBytes() int64 {
	var total int64
	for _, e := range s.docs {
		total += int64(e.Size)
	}
	for _, n := range s.assets {

	}

import (

// uploadAsset 接收一张图片（multipart 字段 file），按内容入库后返回访问地址。
// 大小与类型校验都在 store.SaveImage 里，这里只负责搬运与错误映射。

func imageContentType(name string) string {
	switch {
	case strings.HasSuffix(name, ".png"):
		return "image/png"
	case strings.HasSuffix(name, ".jpg"):
		return "image/jpeg"
	case strings.HasSuffix(name, ".gif"):

	}
	return "image/webp"

// AssetHandler 提供匿名图片读取，挂在 /a/ 前缀下、不经过 withAuth。
// 用户名必须在注册表里、文件名必须是内容哈希格式，否则一律 404。
func AssetHandler(reg *users.Registry) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		rest := strings.TrimPrefix(r.URL.Path, "/a/")
		user, name, ok := strings.Cut(rest, "/")
		if !ok || user == "" || strings.Contains(name, "/") {
			http.NotFound(w, r)
			return
		}
		st, err := reg.Store(user)
		if err != nil {
			http.NotFound(w, r)
			return
		}
		f, fi, err := st.OpenImage(name)
		if err != nil {
			http.NotFound(w, r)
			return
		}
		defer f.Close()
		w.Header().Set("Content-Type", imageContentType(name))
		// 文件名就是内容哈希，内容永不变，可以长缓存。
		w.Header().Set("Cache-Control", "public, max-age=31536000, immutable")
		http.ServeContent(w, r, name, fi.ModTime(), f)
	})
}

// imageBytes 解析 upload_image 的两种入参：source_path（仅 stdio）或 data_base64。
func imageBytes(srcPath, b64 string, allowPath bool) ([]byte, error) {
	switch {
	case srcPath != "" && b64 != "":
		return nil, errors.New("provide either source_path or data_base64, not both")
	case srcPath != "":
		if !allowPath {
			return nil, errors.New("source_path is only available over stdio; use data_base64 for HTTP")
		}
		return os.ReadFile(srcPath)
	case b64 != "":
		if i := strings.Index(b64, ","); strings.HasPrefix(b64, "data:") && i > 0 {
			b64 = b64[i+1:] // 容忍 data:image/png;base64, 前缀
		}
		data, err := base64.StdEncoding.DecodeString(b64)
		if err != nil {
			return nil, errors.New("data_base64 is not valid base64")
		}
		return data, nil
	}
	return nil, errors.New("provide data_base64 (or source_path over stdio)")
}

  /* ---------- 图片：粘贴 / 拖拽 / 选择后上传，按光标位置插入 ---------- */

  // clipboardImages 取剪贴板里的图片文件；Chrome/Firefox 走 files，Safari 走 items。
  function clipboardImages(e) {
    var dt = e.clipboardData;
    if (!dt) { return []; }
    if (dt.files && dt.files.length) { return Array.prototype.slice.call(dt.files); }
    var out = [];
    Array.prototype.forEach.call(dt.items || [], function (it) {
      if (it.kind === 'file') { var f = it.getAsFile(); if (f) { out.push(f); } }
    });
    return out;
  }

  function altOf(f) {
    var n = (f.name || '').replace(/\.[^.]+$/, '').replace(/[\[\]]/g, ' ').trim();
    return n || I18N.t('img.alt');
  }

  function insertAtCursor(ta, text) {
    var start = ta.selectionStart, end = ta.selectionEnd, v = ta.value;
    ta.value = v.slice(0, start) + text + v.slice(end);
    var pos = start + text.length;
    ta.setSelectionRange(pos, pos);
    ta.dispatchEvent(new Event('input', { bubbles: true }));
  }

  // insertImages 串行上传，逐张按选择顺序插入；单张失败不中断其余。
  function insertImages(ta, files, msgId) {
    var list = Array.prototype.filter.call(files || [], function (f) { return /^image\//.test(f.type); });
    if (!list.length) { return; }
    var failed = 0;
    function step(i) {
      if (i >= list.length) {
        if (failed) { msg(msgId, I18N.t('img.upload_failed', { n: failed }), true); }
        else { msg(msgId, ''); }
        return;
      }
      var f = list[i];
      msg(msgId, I18N.t('img.uploading', { done: i + 1, total: list.length }));
      var fd = new FormData();
      fd.append('file', f, f.name || 'image');
      api('/api/assets', { method: 'POST', body: fd }).then(function (d) {
        insertAtCursor(ta, '![' + altOf(f) + '](' + d.url + ')\n');
        step(i + 1);
      }).catch(function () {
        failed++;
        step(i + 1);
      });
    }
    step(0);
  }

  // bindImageInput 给一个 textarea 接上粘贴与拖拽；按钮单独绑。
  function bindImageInput(ta, msgId) {
    ta.addEventListener('paste', function (e) {
      var files = clipboardImages(e);
      if (!files.length) { return; }
      e.preventDefault();
      insertImages(ta, files, msgId);
    });
    ta.addEventListener('dragover', function (e) {
      if (e.dataTransfer && e.dataTransfer.types && Array.prototype.indexOf.call(e.dataTransfer.types, 'Files') >= 0) {
        e.preventDefault();
      }
    });
    ta.addEventListener('drop', function (e) {
      if (!e.dataTransfer || !e.dataTransfer.files || !e.dataTransfer.files.length) { return; }
      e.preventDefault();
      insertImages(ta, e.dataTransfer.files, msgId);
    });
  }
```

    var pending = imgs.map(function (img) {
      if (img.complete) { return Promise.resolve(); }
      return new Promise(function (resolve) {
        img.addEventListener('load', resolve, { once: true });
        img.addEventListener('error', resolve, { once: true });
      });
    });
    return Promise.race([
      Promise.all(pending),
      new Promise(function (resolve) { setTimeout(resolve, timeoutMs || 5000); })
    ]);
  }
```
