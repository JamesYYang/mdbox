# 多语言（中 / 英）支持 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 给 mdbox 增加中 / 英双语支持——浏览器界面、后端错误、MCP 工具文案全部可切换到英文，并在顶栏提供一个语言切换图标。

**Architecture:** 前端新增 `web/i18n.js`（零依赖字典 + `data-i18n` 静态重写），语言存 `localStorage['mdbox:lang']`，默认按 `navigator.language` 识别；所有 `fetch` 带 `Accept-Language`。后端新增 `internal/i18n` 消息目录，`users` / `store` 的哨兵错误改为「键」，`api.writeErr` 按请求语言翻译。MCP 工具文案固定英文。

**Tech Stack:** Go 1.23（标准库 `net/http`、`context`）、vanilla JS（无构建、无 CDN）、`github.com/mark3labs/mcp-go`（既有依赖）。

设计稿：`docs/superpowers/specs/2026-10-04-i18n-design.md`

## Global Constraints

- 语言代码只用 `zh` / `en`（不带地区）；后端无法识别时默认 `en`。
- 文案键命名 `<域>.<项>`，全小写、点号分隔；中英字典键集合必须一一对应。
- 前端静态文案只有**叶子元素**才加 `data-i18n*`（会写 `textContent`，不能覆盖子元素）；混合文本需单独包一层 `<span>`。
- MCP 工具描述 / 参数 / 结果固定**英文**；工具名（`list_docs` 等）是标识符，不动。
- **不翻译**部署者视角字符串：`main.go` 的 CLI flag 帮助与日志、`internal/config` 生成的 `config.yaml` 注释。
- 代码注释与不参与本地化的字符串保持中文（仓库约定）。
- 不新增任何第三方依赖。
- 前端经 `//go:embed web` 打包，`web/` 改动必须 `go build` 后才生效；`*.js` 在 `.gitattributes` 标记为 `binary`，diff 不显示属正常。
- 构建门禁（每个后端任务结束跑）：`gofmt -l <改动的 go 文件>`、`go vet ./...`、`go test ./...`、`go build -o /tmp/mdbox-build .`。
- 浏览器验收由作者本人做，不写自动化前端测试。

```go
// Package i18n 提供后端文案的中英文目录与请求语言解析。
// 只覆盖浏览器可见的错误文案；CLI / 日志等部署者视角的字符串不在此列。
package i18n

import "strings"

// Lang 是受支持的语言代码。
type Lang string

const (
	ZH Lang = "zh"
	EN Lang = "en"

	// Default 是无法识别语言时的兜底（面向英文推广）。
	Default = EN
)

// Key 是文案键，命名形如 "<域>.<项>"。前后端共用同一套键名。
type Key string

const (
	KeyUnauthorized      Key = "error.unauthorized"
	KeyInternal          Key = "error.internal"
	KeyBadJSON           Key = "error.bad_json"
	KeyBadForm           Key = "error.bad_form"
	KeyBadCredentials    Key = "auth.bad_credentials"
	KeyRegisterClosed    Key = "auth.register_closed"
	KeyRegisterLimited   Key = "auth.register_limited"
	KeyShareGone         Key = "share.gone"
	KeyUserNameInvalid   Key = "user.name_invalid"
	KeyUserPasswordWeak  Key = "user.password_weak"
	KeyUserExists        Key = "user.exists"
	KeyUserNotFound      Key = "user.not_found"
	KeyUserPasswordWrong Key = "user.password_wrong"
	KeyDocNotFound       Key = "doc.not_found"
)

var catalog = map[Lang]map[Key]string{
	EN: {
		KeyUnauthorized:      "Sign-in required",
		KeyInternal:          "Internal server error",
		KeyBadJSON:           "Invalid request",
		KeyBadForm:           "Invalid request",
		KeyBadCredentials:    "Incorrect username or password",
		KeyRegisterClosed:    "Registration is closed",
		KeyRegisterLimited:   "Too many attempts, please try again later",
		KeyShareGone:         "This share link is invalid or has expired",
		KeyUserNameInvalid:   "Username may contain only lowercase letters, digits, underscores and hyphens, 3-32 characters",
		KeyUserPasswordWeak:  "Password must be 8-72 characters",
		KeyUserExists:        "Username is already taken",
		KeyUserNotFound:      "User not found",
		KeyUserPasswordWrong: "Current password is incorrect",
		KeyDocNotFound:       "Document not found",
	},
	ZH: {
		KeyUnauthorized:      "需要登录",
		KeyInternal:          "服务器内部错误",
		KeyBadJSON:           "请求格式错误",
		KeyBadForm:           "请求格式错误",
		KeyBadCredentials:    "用户名或密码错误",
		KeyRegisterClosed:    "注册已关闭",
		KeyRegisterLimited:   "注册过于频繁，请稍后再试",
		KeyShareGone:         "分享不存在或已失效",
		KeyUserNameInvalid:   "用户名只能包含小写字母、数字、下划线和连字符，长度 3-32",
		KeyUserPasswordWeak:  "密码长度需要在 8-72 个字符之间",
		KeyUserExists:        "用户名已被占用",
		KeyUserNotFound:      "用户不存在",
		KeyUserPasswordWrong: "旧密码不正确",
		KeyDocNotFound:       "文档不存在",
	},
}

// T 返回给定语言下键对应的文案；缺翻译回退英文，再回退键名本身。
func T(l Lang, k Key) string {
	if s, ok := catalog[l][k]; ok {
		return s
	}
	if s, ok := catalog[Default][k]; ok {
		return s
	}
	return string(k)
}

// Known 判断 s 是否为已知文案键（供错误输出决定是否翻译）。
func Known(s string) bool {
	_, ok := catalog[Default][Key(s)]
	return ok
}

// FromAcceptLanguage 取 Accept-Language 中第一个可识别语言：前缀 zh → 中文，
// 前缀 en → 英文，都不命中（如 fr）继续往后找；全部未命中则默认英文。
func FromAcceptLanguage(h string) Lang {
	for _, part := range strings.Split(h, ",") {
		tag := strings.TrimSpace(strings.SplitN(part, ";", 2)[0])
		tag = strings.ToLower(tag)
		switch {
		case strings.HasPrefix(tag, "zh"):
			return ZH
		case strings.HasPrefix(tag, "en"):
			return EN
		}
	}
	return Default
}
```

```go
func (s *Server) withAuth(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		ctx := context.WithValue(r.Context(), ctxLang, i18n.FromAcceptLanguage(r.Header.Get("Accept-Language")))
		r = r.WithContext(ctx)
		if publicPath(r.URL.Path) {
			next.ServeHTTP(w, r)
			return

			return

			return
		}
		ctx = context.WithValue(r.Context(), ctxUser, name)
		ctx = context.WithValue(ctx, ctxStore, st)
		next.ServeHTTP(w, r.WithContext(ctx))
	})
```

```go
// localize 把已知文案键翻成当前请求语言，其余原样返回。
func localize(r *http.Request, msg string) string {
	if i18n.Known(msg) {
		return i18n.T(langOf(r), i18n.Key(msg))
	}
	return msg
}

func writeErr(w http.ResponseWriter, r *http.Request, status int, msg string) {
	writeJSON(w, status, map[string]string{"error": localize(r, msg)})
}

// writeBadReq 返回本地化的「请求格式错误」，原始错误只写日志、不回给客户端。
func writeBadReq(w http.ResponseWriter, r *http.Request, key i18n.Key, err error) {
	log.Printf("api %s %s: %v", r.Method, r.URL.Path, err)
	writeErr(w, r, http.StatusBadRequest, string(key))
}
```

```js
  function detect() {
    var saved = null;
    try { saved = localStorage.getItem('mdbox:lang'); } catch (e) { /* ignore */ }
    if (saved === 'zh' || saved === 'en') { return saved; }
    var nav = (navigator.language || navigator.userLanguage || '').toLowerCase();
    return nav.indexOf('zh') === 0 ? 'zh' : 'en';
  }

  function t(key, vars) {
    var s = (DICT[current] || DICT[DEFAULT])[key];
    if (s == null) { s = DICT[DEFAULT][key]; }
    if (s == null) { return key; }
    if (vars) {
      s = s.replace(/\{(\w+)\}/g, function (m, name) {
        return vars[name] != null ? vars[name] : m;
      });
    }
    return s;
  }

  function lang() { return current; }

  function setLang(l) {
    if (l !== 'zh' && l !== 'en') { return; }
    current = l;
    try { localStorage.setItem('mdbox:lang', l); } catch (e) { /* ignore */ }

  // applyStatic 把 [data-i18n*] 属性按当前语言写回 DOM，并同步 <html lang>。
  // 只处理叶子元素（写 textContent 会覆盖子节点）；混合文本需在 HTML 里单独包一层。
  function applyStatic(root) {
    root = root || document;
    var map = {
      'data-i18n': 'textContent',
      'data-i18n-title': 'title',
      'data-i18n-placeholder': 'placeholder',
      'data-i18n-aria': 'aria-label'
    };
    Object.keys(map).forEach(function (attr) {
      var prop = map[attr];
      var nodes = root.querySelectorAll('[' + attr + ']');
      for (var i = 0; i < nodes.length; i++) {
        var v = t(nodes[i].getAttribute(attr));
        if (prop === 'textContent') { nodes[i].textContent = v; }
        else { nodes[i].setAttribute(prop, v); }
      }
    });
    document.documentElement.setAttribute('lang', current === 'zh' ? 'zh-CN' : 'en');
  }

  current = detect();

  global.I18N = {
    t: t,
    lang: lang,
    setLang: setLang,
    applyStatic: applyStatic,
    detect: detect
  };
})(window);
```

```js
  function api(path, opts) {
    opts = opts || {};
    var init = { method: opts.method || 'GET' };
    if (opts.body) { init.body = opts.body; }
    init.headers = opts.headers || {};
    init.headers['Accept-Language'] = I18N.lang();
    return fetch(path, init).then(function (res) {

      });
    });
  }
```

- [ ] **Step 3: 新增语言相关函数**

在 `/* ---------- 基础工具 ---------- */` 附近（或 toggleTheme 之后）加入：

```js
  /* ---------- 语言 ---------- */

  var refreshLoginMode = null; // 由 bindLogin 注入：切语言时按当前登录/注册态刷新提示

  function refreshTitle() {
    if (shareCtx) {
      document.title = (shareCtx.title || I18N.t('share.title')) + ' · MDBox';
    } else {
      document.title = I18N.t('app.title');
    }
  }

  function updateLangBtn() {
    var btn = $('lang-btn');
    if (!btn) { return; }
    var label = I18N.lang() === 'zh' ? 'Switch to English' : '切换到中文';
    btn.title = label;
    btn.setAttribute('aria-label', label);
  }

  // applyLang 在切换语言后重渲染当前视图；不触碰编辑态输入内容。
  function applyLang() {
    I18N.applyStatic();
    refreshTitle();
    updateLangBtn();
    if (!$('view-login').hidden) {
      if (refreshLoginMode) { refreshLoginMode(); }
      return;
    }
    if (state.user) {
      loadSidebar();
      if (state.mode === 'list') {
        renderListHead();
        renderList();
      } else if (state.mode === 'preview' && state.current) {
        $('d-meta').textContent = metaText(state.current);
      }
    }
  }

  function bindLangButton() {
    $('lang-btn').addEventListener('click', function () {
      I18N.setLang(I18N.lang() === 'zh' ? 'en' : 'zh');
      applyLang();
    });
    updateLangBtn();
  }
```
